24

29
30            for (auto& di : dir) {
31                int nr = r0 + di[0], nc = c0 + di[1];
32                if (nr >= 0 && nc >= 0 && nr < row && nc < col) {
33                    int nid = nr * col + nc;
34                    if (land[nid]) unite(id, nid);
35                }
36            }
37
38            if (r0 == 0) unite(id, top);
39            if (r0 == row - 1) unite(id, bottom);
40
41            if (find(top) == find(bottom)) return d;
42        }
43        return 0;
44    }
