1/**
2 * Definition for a binary tree node.
3 * struct TreeNode {
4 *     int val;
5 *     TreeNode *left;
6 *     TreeNode *right;
7 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
8 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
9 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
10 * };
11 */
12class Solution {
13public:
14    long long total = 0, best = 0;
15    static const int MOD = 1000000007;
16
17    long long sum(TreeNode* root) {
18        if (!root) return 0;
19        return root->val + sum(root->left) + sum(root->right);
20    }
21
22    long long dfs(TreeNode* root) {
23        if (!root) return 0;
24        long long s = root->val + dfs(root->left) + dfs(root->right);
25        best = max(best, s * (total - s));
26        return s;
27    }
28
29    int maxProduct(TreeNode* root) {
30        total = sum(root);
31        dfs(root);
32        return best % MOD;
33    }
34};
