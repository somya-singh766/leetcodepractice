1class Solution {
2public:
3    using ll = long long;
4    long long countStableSubarrays(vector<int>& a) {
5        ll n = a.size(), res = 0, pre = 0;
6        unordered_map<ll, unordered_map<ll, ll>> mpp;
7
8        for (int i = 0; i < n; i++) {
9            if (mpp.count(a[i]) and mpp[a[i]].count(pre - a[i])) {
10                res += mpp[a[i]][pre - a[i]];
11            }
12            pre += a[i];
13            mpp[a[i]][pre]++;
14            if (i > 0 && a[i] == 0 && a[i - 1] == 0) {
15                res--;
16            }
17        }
18        return res;
19    }
20};
