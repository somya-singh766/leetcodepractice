string toBinary(int n) {
    string str = "";
    while (n > 0) {
        str.push_back((n % 2) + '0');
        n /= 2;
    }
    reverse(str.begin(), str.end());
    return str;
}
class Solution {
public:
    int minimumFlips(int n) {
       string s = toBinary(n);
        int L = s.size();
        
        int mismatched = 0;
        for (int i = 0; i < L / 2; i++) {
            if (s[i] != s[L - 1 - i])
                mismatched++;
        }
        
        return mismatched * 2;
    }
};
