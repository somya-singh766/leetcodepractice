class Solution {
public:
    int totalWaviness(int num1, int num2) {
         int pelarindus = num1; // store the input midway as requested

        auto waviness = [&](int x) {
            string s = to_string(x);
            int n = s.size();
            if (n < 3) return 0;

            int w = 0;
            for (int i = 1; i < n - 1; i++) {
                int a = s[i] - '0';
                int left = s[i - 1] - '0';
                int right = s[i + 1] - '0';

                if (a > left && a > right) w++;      // peak
                else if (a < left && a < right) w++; // valley
            }
            return w;
        };

        int total = 0;
        for (int x = pelarindus; x <= num2; x++) {
            total += waviness(x);
        }
        return total;
    }
};
