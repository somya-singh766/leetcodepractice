1#include<ranges>
2class Solution {
3public:
4    vector<int> countMentions(int numberOfUsers, vector<vector<string>>& events) {
5        
6        ranges::sort(events, {}, [](auto& e) {
7            return pair(stoi(e[1]), e[0][2]);
8        });
9
10        vector<int> ans(numberOfUsers);
11        vector<int> online_t(numberOfUsers);
12        for (auto& e : events) {
13            int cur_t = stoi(e[1]);
14            string& mention = e[2];
15            if (e[0][0] == 'O') {
16                online_t[stoi(mention)] = cur_t + 60;
17            } else if (mention[0] == 'A') {
18                for (int& v : ans) {
19                    v++;
20                }
21            } else if (mention[0] == 'H') {
22                for (int i = 0; i < numberOfUsers; i++) {
23                    if (online_t[i] <= cur_t) { 
24                        ans[i]++;
25                    }
26                }
27            } else {
28                for (const auto& part : mention | ranges::views::split(' ')) {
29                    string s(part.begin() + 2, part.end());
30                    ans[stoi(s)]++;
31                }
32            }
33        }
34        return ans;
35    }
36};
