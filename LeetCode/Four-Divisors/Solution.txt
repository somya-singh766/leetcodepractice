1class Solution {

12    int sumOne(int n) {
13        int p = round(cbrt(n));
14        if ((long long)p * p * p == n && isPrime(p)) {
15            return 1 + p + p*p + p*p*p;
16        }
17

19            if (n % i == 0) {

21                if (a != b && isPrime(a) && isPrime(b)) {
22                    return 1 + a + b + n;
23                }

25            }
26        }

28    }

30    bool isPrime(int x) {
31        if (x < 2) return false;

33            if (x % i == 0) return false;
34        }
35        return true;
36    }
