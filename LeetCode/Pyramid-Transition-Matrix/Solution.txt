3    vector<string> gen_neighbor(const string& node,
4                                unordered_map<string, unordered_set<char>>& mp) {
5        vector<string> res{""};
6        int n = node.size();
7
8        for (int i = 1; i < n; i++) {
9            string key;
10            key.push_back(node[i - 1]);
11            key.push_back(node[i]);
12
13            if (mp.count(key) == 0) {
14                return {};
15            }
16
17            vector<string> newRes;
18            for (const string& a : res) {
19                for (char c : mp[key]) {
20                    newRes.push_back(a + c);
21                }
22            }
23            res.swap(newRes);
24        }
25        return res;
26    }
27
28    bool pyramidTransition(string bottom, vector<string>& allowed) {
29        unordered_map<string, unordered_set<char>> mp;
30        for (const string& s : allowed) {
31            string key;
32            key.push_back(s[0]);
33            key.push_back(s[1]);
34            mp[key].insert(s[2]);
35        }
36
37        unordered_set<string> visited;
38
39        function<bool(const string&)> dfs = [&](const string& node) -> bool {
40            if (node.size() == 1) {
41                return true;
42            }
43            if (visited.count(node)) {
44                return false;
45            }
46
47            for (const string& nxt : gen_neighbor(node, mp)) {
48                if (dfs(nxt)) {
49                    return true;
50                }
51            }
52
53            visited.insert(node);
54            return false;
55        };
56
57        return dfs(bottom);
58    }
