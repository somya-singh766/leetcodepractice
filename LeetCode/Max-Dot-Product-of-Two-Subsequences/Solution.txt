1class Solution {
2public:
3    static int maxDotProduct(vector<int>& nums1, vector<int>& nums2) {
4        const int n1=nums1.size(), n2=nums2.size();
5        if (n1<n2) // for less space, interchange nums1 & nums2
6            return maxDotProduct(nums2, nums1);
7        int dp[2][501];
8        fill(&dp[0][0], &dp[0][0]+2*501, INT_MIN);
9        int res=INT_MIN;
10        for(int i=n1-1; i>=0; i--){
11            for(int j=n2-1; j>=0; j--){
12                int x=nums1[i]*nums2[j], tmp=dp[i&1][j];
13                tmp=max(tmp, x);
14                tmp=max(tmp, x+(i+1<n1 & j+1<n2 ? dp[(i+1)&1][j+1]:0));
15                tmp=max(tmp, dp[i&1][(j+1)]);
16                dp[i&1][j]=max(tmp, dp[(i+1)&1][j]);
17                res=max(res, dp[i&1][j]);
18            }
19        }
20        return res;
21    }
22};
